set(_dotnet_bindings_dir "${CMAKE_CURRENT_BINARY_DIR}/dotnet-bindings")
set(_dotnet_build_cmd
    dotnet build
    --configuration $<IF:$<CONFIG:Debug>,Debug,Release>
    \"${CMAKE_CURRENT_SOURCE_DIR}/Oblo.Managed.csproj\"
    -o \"$<TARGET_FILE_DIR:oblo::dotnet::runtime>/managed\"
    "-p:BaseIntermediateOutputPath=${CMAKE_CURRENT_BINARY_DIR}/obj/"
    "-p:IntermediateOutputPath=${CMAKE_CURRENT_BINARY_DIR}/obj/$<IF:$<CONFIG:Debug>,Debug,Release>/"
    "-p:GeneratedDotNetDir=${_dotnet_bindings_dir}"
)

set(_dotnet_generated_cs "${_dotnet_bindings_dir}/Bindings.gen.cs")

define_property(GLOBAL PROPERTY oblo_dotnet_bindings_generated_source BRIEF_DOCS "The .NET bindings source file generated by obindgen")
set_property(GLOBAL PROPERTY oblo_dotnet_bindings_generated_source ${_dotnet_generated_cs})

add_custom_target(
    Oblo.Managed-build
    ALL
    COMMAND ${_dotnet_build_cmd}
    COMMENT "Building .NET code: ${_dotnet_build_cmd}"
    DEPENDS "${_dotnet_generated_cs}"
)

add_custom_command(
    OUTPUT ${_dotnet_generated_cs}
    COMMAND $<TARGET_FILE:obindgen> "${_dotnet_generated_cs}"
    # DEPENDS oblo::graphics
    DEPENDS $<TARGET_PROPERTY:oblo::cmake::meta,reflection_projects>
)

target_sources(Oblo.Managed-build PRIVATE ${_dotnet_generated_cs})

set_target_properties(
    Oblo.Managed-build
    PROPERTIES
    PROJECT_LABEL ".NET Build"
)

oblo_set_target_folder(Oblo.Managed-build build)

if(OBLO_GENERATE_CSHARP)
    include_external_msproject(Oblo.Managed "${CMAKE_CURRENT_SOURCE_DIR}/Oblo.Managed.csproj")

    # We add the csproj for convenience, although it would fail to build since it's missing the reference to the generated code
    # Nonetheless it's nice to have in the solution when developing, we just need to exclude it from the ALL target
    set_target_properties(Oblo.Managed PROPERTIES EXCLUDE_FROM_ALL TRUE)

    oblo_set_target_folder(Oblo.Managed libraries)
endif()
